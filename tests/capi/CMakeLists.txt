cmake_minimum_required(VERSION 3.20)
project(TessSTEPConsumer LANGUAGES C)
if(NOT TESSSTEP_C_ONLY)
    enable_language(CXX)
endif()
find_package(TessSTEP 0.1 CONFIG REQUIRED)
enable_testing()
add_executable(c_consumer consumer.c)
set_target_properties(c_consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(c_consumer PRIVATE TessSTEP::TessSTEP)
set(TS_CONSUMERS c_consumer)
if(NOT TESSSTEP_C_ONLY)
    find_package(Threads REQUIRED)
    add_executable(cpp_consumer consumer.cpp)
    target_link_libraries(cpp_consumer PRIVATE TessSTEP::TessSTEP Threads::Threads)
    list(APPEND TS_CONSUMERS cpp_consumer)
endif()
foreach(app ${TS_CONSUMERS})
    if(MSVC)
        target_compile_options(${app} PRIVATE /W4 /WX /UNDEBUG)
    else()
        target_compile_options(${app} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    endif()
    if(TESSSTEP_SANITIZERS AND NOT MSVC)
        target_compile_options(${app} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${app} PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME ${app} COMMAND ${app})
endforeach()
